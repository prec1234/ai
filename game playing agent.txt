board = [' ' for i in range(9)]

def display():
    for i in range(0, 9, 3):
        print(board[i], "|", board[i+1], "|", board[i+2])
        print("---------")

def check():
    wins = [(0,1,2), (3,4,5), (6,7,8),
            (0,3,6), (1,4,7), (2,5,8),
            (0,4,8), (2,4,6)]

    for a, b, c in wins:
        if board[a] == board[b] == board[c] != ' ':
            return board[a]

    if ' ' not in board:
        return 'Draw'

    return None

def minimax(is_max):
    result = check()

    if result == 'X':
        return 1
    if result == 'O':
        return -1
    if result == 'Draw':
        return 0

    if is_max:
        best = -100
        for i in range(9):
            if board[i] == ' ':
                board[i] = 'X'
                score = minimax(False)
                board[i] = ' '
                best = max(best, score)
        return best

    else:
        best = 100
        for i in range(9):
            if board[i] == ' ':
                board[i] = 'O'
                score = minimax(True)
                board[i] = ' '
                best = min(best, score)
        return best

def computer():
    best = -100
    move = -1

    for i in range(9):
        if board[i] == ' ':
            board[i] = 'X'
            score = minimax(False)
            board[i] = ' '

            if score > best:
                best = score
                move = i

    board[move] = 'X'

print("Tic Tac Toe Game")
print("You are O and Computer is X")

while True:
    display()

    pos = int(input("Enter position (1-9): ")) - 1

    if pos < 0 or pos > 8 or board[pos] != ' ':
        print("Invalid position!")
        continue

    board[pos] = 'O'

    if check():
        break

    computer()

    if check():
        break

display()

result = check()

if result == 'X':
    print("Computer Wins!")
elif result == 'O':
    print("You Win!")
else:
    print("Game Draw!")
